<!-- Script para la carga automática de partidos semanales -->

<script>

// Este script buscará un archivo 'partidos.json' publicado automáticamente cada semana

fetch('partidos.json')

.then(response => response.json())

.then(data => {

for (const [pairId, match] of Object.entries(data)) {

const card = document.getElementById(pairId);

if (card) {

card.querySelector('.rival').textContent = match.rival;

card.querySelector('.fronton').textContent = match.fronton;

card.querySelector('.horario').textContent = match.fecha_hora;

const badge = card.querySelector('.badge-pending');

if (badge) {

badge.textContent = "Partida Zehaztuta";

badge.className = "badge-match";

}

}

}

})

.catch(err => console.log('Cartelera semanal aún no cargada.'));

</script>
